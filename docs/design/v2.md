/// 检索器主 trait
pub trait Retriever: Send + Sync {
    fn retrieve(
        &self,
        tree: &VectorlessTree,
        query: &str,
        options: &RetrieveOptions,
    ) -> impl Future<Output = Result<RetrievalResult>>;
}

/// 检索策略 trait
pub trait RetrievalStrategy: Send + Sync {
    /// 策略名称
    fn name(&self) -> &str;
    
    /// 执行检索
    fn retrieve(
        &self,
        tree: &VectorlessTree,
        query: &str,
        options: &RetrieveOptions,
    ) -> impl Future<Output = Result<Vec<NodeId>>>;
    
    /// 是否需要 LLM
    fn requires_llm(&self) -> bool;
}

/// 搜索算法 trait
pub trait SearchAlgorithm: Send + Sync {
    /// 搜索相关节点
    fn search(
        &self,
        tree: &VectorlessTree,
        query: &str,
        scorer: &dyn NodeScorer,
        options: &SearchOptions,
    ) -> Vec<SearchPath>;
}

/// 节点评分器 trait
pub trait NodeScorer: Send + Sync {
    /// 计算节点相关性分数 (0.0 - 1.0)
    fn score(&self, node: &TreeNode, query: &str) -> f32;
}

/// 充分性判断 trait
pub trait SufficiencyChecker: Send + Sync {
    /// 判断当前内容是否足够回答问题
    fn check(
        &self,
        content: &str,
        query: &str,
    ) -> impl Future<Output = SufficiencyLevel>;
}

/// 复杂度检测器 trait
pub trait ComplexityDetector: Send + Sync {
    /// 检测查询复杂度
    fn detect(&self, query: &str, tree: &VectorlessTree) -> QueryComplexity;
}

// ==================== 查询复杂度 ====================
pub enum QueryComplexity {
    Simple,      // 关键词可解
    Medium,      // 需要语义理解
    Complex,     // 需要 LLM 深度推理
}

// ==================== 检索选项 ====================
pub struct RetrieveOptions {
    pub top_k: usize,                    // 返回节点数
    pub beam_width: usize,               // Beam Search 宽度
    pub max_iterations: usize,           // 最大迭代次数
    pub include_content: bool,           // 包含内容
    pub include_summaries: bool,         // 包含摘要
    pub strategy: StrategyPreference,    // 策略偏好
    pub sufficiency_check: bool,         // 启用充分性判断
}

pub enum StrategyPreference {
    Auto,                  // 自适应选择
    ForceKeyword,          // 强制关键词
    ForceSemantic,         // 强制语义
    ForceLlm,              // 强制 LLM
}

// ==================== 检索结果 ====================
pub struct RetrievalResult {
    pub nodes: Vec<NodeId>,              // 相关节点
    pub content: String,                 // 聚合内容
    pub score: f32,                      // 总体置信度
    pub is_sufficient: bool,             // 是否充分
    pub strategy_used: String,           // 使用的策略
    pub trace: Vec<NavigationStep>,      // 搜索路径
    pub tokens_used: usize,              // Token 消耗
}

pub struct NavigationStep {
    pub node_id: NodeId,
    pub node_title: String,
    pub score: f32,
    pub decision: NavigationDecision,
    pub depth: usize,
}

// ==================== 充分性级别 ====================
pub enum SufficiencyLevel {
    Sufficient,           // 足够，停止
    PartialSufficient,    // 部分足够，可继续
    Insufficient,         // 不足够，继续
}

// ==================== 搜索路径 ====================
pub struct SearchPath {
    pub nodes: Vec<NodeId>,              // 路径上的节点
    pub score: f32,                      // 路径总分
    pub leaf: NodeId,                    // 叶节点
}
